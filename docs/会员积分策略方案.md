## 七、积分消费公式汇总

### 7.1 计算公式

```javascript
// 积分消耗计算（基于火山引擎官方定价 + 50%服务费）
const POINTS_RATE = {
  // 文本生成：基础 2 积分，超长按 token 计算
  TEXT_BASE: 2,
  TEXT_PER_2K_TOKENS: 1,  // 每 2000 tokens 额外 +1 积分
  
  // 图片生成：17 积分/张（成本¥0.22 × 1.5 ÷ 0.02 = 16.5 → 17）
  IMAGE_PER_ITEM: 17,
  
  // 视频生成：72 积分/秒（成本¥0.95 × 1.5 ÷ 0.02 = 71.25 → 72）
  VIDEO_PER_SECOND: 72,
};

function calculatePoints(type, params) {
  switch (type) {
    case 'TEXT':
      const tokens = params.inputTokens + params.outputTokens;
      return Math.max(2, Math.ceil(tokens / 2000) + 1);
      
    case 'IMAGE':
      return params.count * POINTS_RATE.IMAGE_PER_ITEM;
      
    case 'VIDEO':
      return params.seconds * POINTS_RATE.VIDEO_PER_SECOND;
  }
}
```

### 8.2 积分服务核心逻辑

```javascript
// services/pointsService.js

const POINTS_RATE = {
  TEXT_BASE: 2,
  TEXT_PER_2K_TOKENS: 1,
  IMAGE_PER_ITEM: 17,
  VIDEO_PER_SECOND: 72,
};

/**
 * 计算 AI 任务的积分消耗
 */
function calculatePointsCost(modelCategory, usage, options = {}) {
  switch (modelCategory) {
    case 'TEXT': {
      const totalTokens = (usage.inputTokens || 0) + (usage.outputTokens || 0);
      if (totalTokens <= 2000) return POINTS_RATE.TEXT_BASE;
      return Math.ceil(totalTokens / 2000) + 1;
    }
    case 'IMAGE': {
      const count = usage.itemCount || 1;
      return count * POINTS_RATE.IMAGE_PER_ITEM;  // 17 积分/张
    }
    case 'VIDEO': {
      const seconds = usage.durationSeconds || 5;
      return seconds * POINTS_RATE.VIDEO_PER_SECOND;  // 72 积分/秒
    }
    default:
      return 0;
  }
}

/**
 * 扣除积分（优先套餐积分，再充值积分）
 */
async function deductPoints(userId, pointsCost, sourceType, sourceId) {
  const userPoints = await queryOne(
    'SELECT plan_balance, extra_balance FROM user_points WHERE user_id = ?',
    [userId]
  );
  
  if (!userPoints) throw new Error('用户积分账户不存在');
  
  const totalBalance = userPoints.plan_balance + userPoints.extra_balance;
  if (totalBalance < pointsCost) {
    const error = new Error('积分余额不足');
    error.code = 'INSUFFICIENT_POINTS';
    error.required = pointsCost;
    error.available = totalBalance;
    throw error;
  }
  
  // 优先扣套餐积分
  let planDeduct = Math.min(userPoints.plan_balance, pointsCost);
  let extraDeduct = pointsCost - planDeduct;
  
  await execute(
    `UPDATE user_points 
     SET plan_balance = plan_balance - ?, 
         extra_balance = extra_balance - ?
     WHERE user_id = ?`,
    [planDeduct, extraDeduct, userId]
  );
  
  // 记录流水
  if (planDeduct > 0) {
    await insertTransaction(userId, 'CONSUME', -planDeduct, 'PLAN', sourceType, sourceId);
  }
  if (extraDeduct > 0) {
    await insertTransaction(userId, 'CONSUME', -extraDeduct, 'EXTRA', sourceType, sourceId);
  }
  
  return { planDeduct, extraDeduct, totalDeduct: pointsCost };
}

module.exports = {
  POINTS_RATE,
  calculatePointsCost,
  deductPoints,
};
```
